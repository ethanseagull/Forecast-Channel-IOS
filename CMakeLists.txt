# ForecastOS: the Forecast Channel decompilation on SDL3 (Windows now, iOS later).
#
# Builds the doldecomp/forecast sources (channel code + NW4R + the portable
# parts of the RVL SDK) with Clang as a 64-bit program, on top of the SDK
# replacement in port/. Windows: tools/build_windows.bat.
cmake_minimum_required(VERSION 3.24)
project(ForecastOS C CXX)

set(DECOMP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/forecast-decomp" CACHE PATH
    "checkout of doldecomp/forecast with patches/decomp-port.patch applied (tools/setup_decomp.ps1)")
set(PORT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/port")

if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "ForecastOS needs Clang (the decomp relies on GNU extensions)")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "ForecastOS targets 64-bit hosts only")
endif()
if(NOT EXISTS ${DECOMP_DIR}/build/HAFE/include)
    message(FATAL_ERROR "Build the decomp once first (generates its binary tables): "
                        "cd forecast-decomp; py configure.py; py -m ninja")
endif()

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_EXTENSIONS ON)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_EXTENSIONS ON)

find_package(SDL3 REQUIRED CONFIG)

# --- flags for decomp and port code -------------------------------------------

set(FORECAST_DEFS FORECAST_PORT NONMATCHING)
if(WIN32)
    list(APPEND FORECAST_DEFS _CRT_SECURE_NO_WARNINGS _CRT_NONSTDC_NO_WARNINGS NOMINMAX _USE_MATH_DEFINES)
endif()

set(FORECAST_FLAGS
    # CodeWarrior's wchar_t is 16 bits and the channel's text is UTF-16
    # (already the size on Windows; needed on iOS).
    -fshort-wchar
    # Written for CodeWarrior: type punning and signed wrap are relied on.
    -fno-strict-aliasing
    -fwrapv
    -include ${PORT_DIR}/include/port_compat.h
    $<$<COMPILE_LANGUAGE:CXX>:-fno-rtti>
    $<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions>
    # 64-bit: every pointer squeezed into 32 bits must be found and fixed.
    -Werror=pointer-to-int-cast
    -Werror=int-to-pointer-cast
    -Werror=void-pointer-to-int-cast
    -Wno-unknown-pragmas
    -Wno-multichar
    -Wno-unused-value
    -Wno-unused-private-field
    -Wno-microsoft-cast
    -Wno-microsoft-template
    -Wno-microsoft-exception-spec
    -Wno-ignored-attributes
    -Wno-unknown-attributes
    -Wno-deprecated-declarations
)

# --- decomp sources ---------------------------------------------------------------

# Units linked into the original DOL (tools/decomp-linked-units.txt), minus the
# parts the port replaces: MSL, MetroTRK, the PPC runtime, the RVL SDK (except
# the portable modules below), the Opera browser glue (vcmv) and HBM.
file(STRINGS ${CMAKE_CURRENT_SOURCE_DIR}/tools/decomp-linked-units.txt DECOMP_UNITS)
set(DECOMP_SOURCES)
foreach(unit IN LISTS DECOMP_UNITS)
    # Replaced by port/channel/ versions.
    if(unit STREQUAL "HomeButton")
        continue()
    endif()
    if(unit MATCHES "^(MSL_C|MetroTRK|Runtime\\.PPCEABI\\.H|vcmv)/")
        continue()
    endif()
    if(unit MATCHES "^revolution/")
        if(NOT unit MATCHES "^revolution/(MEM|CX|ARC|TPL|GX)/" AND NOT unit STREQUAL "revolution/MTX/mtx44")
            continue()
        endif()
        # The FIFO, synchronisation and performance counters are hardware;
        # port/gx replaces them.
        if(unit MATCHES "^revolution/GX/GX(Fifo|Misc|Perf)$")
            continue()
        endif()
    endif()
    foreach(ext cpp c)
        if(EXISTS ${DECOMP_DIR}/src/${unit}.${ext})
            list(APPEND DECOMP_SOURCES ${DECOMP_DIR}/src/${unit}.${ext})
            break()
        endif()
    endforeach()
endforeach()

# Referenced from vtables Clang emits but CodeWarrior stripped from the DOL,
# so they are missing from the linked-units list.
foreach(unit nw4r/g3d/res/g3d_resanmvis nw4r/g3d/res/g3d_resanmclr nw4r/g3d/res/g3d_resanmtexsrt)
    list(APPEND DECOMP_SOURCES ${DECOMP_DIR}/src/${unit}.cpp)
endforeach()

file(GLOB PORT_SOURCES CONFIGURE_DEPENDS
    ${PORT_DIR}/sdk/*.c
    ${PORT_DIR}/sdk/*.cpp
    ${PORT_DIR}/res/*.cpp
    ${PORT_DIR}/channel/*.cpp
    ${PORT_DIR}/gx/*.c
    ${PORT_DIR}/gx/*.cpp
)

add_executable(forecast ${DECOMP_SOURCES} ${PORT_SOURCES})
target_compile_definitions(forecast PRIVATE ${FORECAST_DEFS})
target_compile_options(forecast PRIVATE ${FORECAST_FLAGS})
target_include_directories(forecast BEFORE PRIVATE ${PORT_DIR}/include)
# MSL lets C sources include <cstring> and friends.
target_compile_options(forecast PRIVATE $<$<COMPILE_LANGUAGE:C>:-I${PORT_DIR}/include_c>)
target_include_directories(forecast PRIVATE
    ${DECOMP_DIR}/include
    ${DECOMP_DIR}/include/channel
    # Binary tables the channel embeds (generated by the decomp's own build).
    ${DECOMP_DIR}/build/HAFE/include
)
target_link_libraries(forecast PRIVATE SDL3::SDL3)
if(WIN32)
    # Report every unresolved symbol with its real name (tools/gen_stubs.py).
    target_link_options(forecast PRIVATE "LINKER:/errorlimit:0" "LINKER:/demangle:no")
endif()

if(WIN32)
    # Keep a console for the log; SDL3.dll next to the executable.
    add_custom_command(TARGET forecast POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL3::SDL3-shared> $<TARGET_FILE_DIR:forecast>)
endif()
